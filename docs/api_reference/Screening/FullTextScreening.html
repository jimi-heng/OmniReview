 96def coerce_pico_payload(pico_input: Any) -> dict[str, Any]:
 97    if pico_input is None:
 98        return {}
 99    if isinstance(pico_input, Mapping):

43def load_pdf_bytes(pdf_path: Path, pages: Optional[list[int]]) -> bytes:
44    if pdf_path.suffix.lower() == ".txt":
45        return pdf_path.read_bytes()
46
47    data = pdf_path.read_bytes()
48    if pages is None:
49        return data
50
51    if fitz is None:
52        raise RuntimeError("PyMuPDF is required for pdf_pages/page slicing. Install `pymupdf`.")
53
54    with fitz.open(stream=data, filetype="pdf") as src:
55        page_count = src.page_count
56        selected = [page for page in pages if 1 <= page <= page_count]
57        if not selected:
58            return data
59
60        with fitz.open() as out:
61            for page in selected:
62                out.insert_pdf(src, from_page=page - 1, to_page=page - 1)
63            return out.write()

def parse_pages_spec(spec: str) -> Optional[list[int]]: 
16def parse_pages_spec(spec: str) -> Optional[list[int]]:
17    normalized = (spec or "").strip().lower()
18    if normalized in {"", "all", "*"}:
19        return None
20
21    pages: set[int] = set()
22    for part in normalized.split(","):
23        part = part.strip()
24        if not part:
25            continue
26        if "-" in part:
27            start_text, end_text = part.split("-", 1)
28            start = int(start_text.strip())
29            end = int(end_text.strip())
30            if start <= 0 or end <= 0:
31                continue
32            if start > end:
33                start, end = end, start
34            pages.update(range(start, end + 1))
35        else:
36            page = int(part)
37            if page > 0:
38                pages.add(page)
39
40    return sorted(pages) if pages else None

 45    """
 46    Process a single paper through the full-text screening pipeline.
 47

 49        1. Load PDF or text input
 50        2. Construct LLM prompt
 51        3. Call model with retry
 52        4. Parse structured output (with fallback retry)

 54
 55    Args:
 56        client (httpx.AsyncClient): Shared HTTP client.
 57        runtime (FullTextScreeningRuntime): Runtime configuration.
 58        request (FullTextRequest): Paper + PICO + metadata.
 59
 60    Returns:
 61        dict[str, Any]:
 62            {
 63                "id": paper_id,
 64                "decision": str,
 65                "evidence": JSON string,
 66                "hl_path": str | None
 67            }
 68    """
 69    paper = request.paper
 70    source_path = paper.source_path
 71
 72    if not source_path.exists():
 73        return error_result(
 74            paper.paper_id,
 75            error_type="missing_file",
 76            message=f"Input file not found: {source_path}",
 77        )
 78
 79    # === Load input ===
 80    try:
 81        pdf_bytes = None
 82        document_text = None
 83
 84        if paper.is_text:
 85            document_text = load_text_content(source_path)
 86        else:
 87            pdf_bytes = load_pdf_bytes(source_path, runtime.selected_pages)
 88
 89    except Exception as exc:
 90        return error_result(
 91            paper.paper_id,
 92            error_type="input_read_error",
 93            message=str(exc),
 94        )
 95
 96    try:
 97        parsed_output: dict[str, Any] | None = None
 98        raw_text = ""
 99
100        # === LLM call + parse retry loop ===
101        for parse_retry in (False, True):
102            prompt = (
103                get_system_prompt(request.project_type)
104                + "\n\n"
105                + build_user_prompt(request, parse_retry=parse_retry)
106            )
107
108            payload = build_payload(
109                runtime.model,
110                prompt=prompt,
111                filename=source_path.name,
112                file_bytes=pdf_bytes,
113                document_text=document_text,
114            )
115
116            response = await call_with_retry(
117                post_with_auth_fallback,
118                client,
119                runtime.endpoint,
120                runtime.api_key,
121                payload,
122                runtime.timeout,
123                retries=runtime.retries,
124                delay_s=runtime.retry_delay,
125            )
126
127            if response.status_code != 200:
128                return error_result(
129                    paper.paper_id,
130                    error_type="http_error",
131                    message=f"Upstream returned status {response.status_code}",
132                )

141
142                if parse_retry:

145                        error_type="parse_error",
146                        message=str(exc),

148                    )
149

154                message="Failed to parse structured output",
155                raw_output=raw_text,
156            )
157
158        # === Extract result ===

161        conclusion = str(parsed_output["reason"])
162

200        )
