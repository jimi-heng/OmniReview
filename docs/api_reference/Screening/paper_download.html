159    def _is_pdf_file_valid(self, path: str, eof_tail_bytes: int = 8192) -> bool:
160        try:

590    def _download_pdf(self, pdf_url, doi, output_dir):
591        base_name = doi.replace("/", "_")
592        pdf_path = os.path.join(output_dir, base_name + ".pdf")
593
594        headers = {
595            "User-Agent": self.session.headers.get("User-Agent", ""),
596            "Accept": "application/pdf,application/octet-stream;q=0.9,*/*;q=0.8",
597            "Accept-Language": "en-US,en;q=0.9",
598        }
599        if self._elsevier_referer and ("sciencedirect" in pdf_url or "elsevier" in pdf_url):
600            headers["Referer"] = self._elsevier_referer
601
602        max_attempts = 3
603
604        for attempt in range(1, max_attempts + 1):
605            try:
606                safe_print(f"  Downloading PDF: {pdf_url} (attempt {attempt}/{max_attempts})")
607                response = self._request(
608                    "GET",
609                    pdf_url,
610                    headers=headers,
611                    stream=True,
612                    allow_redirects=True,
613                )
614
615                if response.status_code in (401, 403):
616                    raise PermissionError(f"HTTP {response.status_code}")
617
618                if response.status_code != 200:
619                    safe_print(f"  ERROR: HTTP {response.status_code} for PDF download")
620                    continue
621
622                content_type = response.headers.get("Content-Type", "").lower()
623                if "pdf" not in content_type and "application/octet-stream" not in content_type:
624                    try:
625                        chunk = next(response.iter_content(chunk_size=4096))
626                    except StopIteration:
627                        safe_print("  ERROR: Empty response content")
628                        continue
629                    if not chunk.startswith(b"%PDF"):
630                        safe_print(f"  ERROR: Non-PDF response (Content-Type: {content_type})")
631                        continue
632
633                    with open(pdf_path, "wb") as f:
634                        f.write(chunk)
635                        for data in response.iter_content(chunk_size=8192):
636                            if data:
637                                f.write(data)
638                else:
639                    with open(pdf_path, "wb") as f:
640                        for chunk in response.iter_content(chunk_size=8192):
641                            if chunk:
642                                f.write(chunk)
643
644                if self._is_pdf_file_valid(pdf_path):
645                    safe_print(f"  SUCCESS: PDF saved to {pdf_path}")
646                    return True
